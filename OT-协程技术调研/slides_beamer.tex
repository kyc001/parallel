\section{问题：协程到底解决什么}

\begin{frame}{一个判断：协程让等待更便宜}
  \begin{columns}[T,totalwidth=\linewidth]
    \begin{column}{0.56\linewidth}
      {\LARGE\bfseries\color{nkuBlue}协程降低的是“等待”的代价，\\不是“计算”的代价。\par}
      \vspace{0.45cm}
      {\color{muted}
      它把回调、阻塞线程和事件循环中的零散状态，还原成一段可以暂停、可以恢复的线性代码；代价没有消失，而是转移到编译器、运行时调度器和 I/O 后端。}
    \end{column}
    \begin{column}{0.38\linewidth}
      \begin{block}{本次汇报回答}
        \begin{enumerate}
          \item 为什么会从进程、线程、事件循环走到协程？
          \item 有栈、无栈协程在底层分别保存什么？
          \item Go、Java、C++、Python 等语言为什么选择不同路线？
          \item 本机实验说明协程擅长什么、不擅长什么？
        \end{enumerate}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{并发模型演进：等待任务越来越多}
  \centering
  \begin{tikzpicture}[

\section{机制：可暂停函数如何工作}

\begin{frame}[fragile]{心智模型：可以中途停下的函数}
  \begin{columns}[T,totalwidth=\linewidth]
    \begin{column}{0.48\linewidth}
      \begin{exampleblock}{普通函数}
\begin{lstlisting}[language=C++]
Resp handle(Req r) {
  auto u = db.query(r.id);
  auto d = rpc.fetch(u);
  return render(d);
}
\end{lstlisting}
      \end{exampleblock}
      {\small\color{muted}调用栈一路向下，直到函数返回；如果 I/O 阻塞，承载它的线程也阻塞。}
    \end{column}
    \begin{column}{0.48\linewidth}
      \begin{block}{协程函数}
\begin{lstlisting}[language=C++]
Task<Resp> handle(Req r) {
  auto u = co_await db.query(r.id);
  auto d = co_await rpc.fetch(u);
  co_return render(d);
}
\end{lstlisting}
      \end{block}
      {\small\color{muted}每个 \texttt{co\_await} 都可能保存状态并交还执行权，之后由调度器恢复。}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{无栈协程：编译器生成状态机}
  \begin{columns}[T,totalwidth=\linewidth]
    \begin{column}{0.50\linewidth}
      \centering
      \begin{tikzpicture}[
        state/.style={draw=lineGray, fill=softBg, rounded corners=2pt, minimum width=2.45cm, minimum height=0.78cm, align=center},
        arr/.style={-{Stealth[length=2.1mm]}, thick, draw=accentBlue}
      ]
        \node[state] (s0) {状态 0\\{\scriptsize 进入函数}};
        \node[state,below=0.65cm of s0] (s1) {状态 1\\{\scriptsize 等待 db.query}};
        \node[state,below=0.65cm of s1] (s2) {状态 2\\{\scriptsize 等待 rpc.fetch}};
        \node[state,below=0.65cm of s2] (s3) {完成\\{\scriptsize co\_return}};
        \draw[arr] (s0) -- node[right,scale=0.72]{首次挂起} (s1);
        \draw[arr] (s1) -- node[right,scale=0.72]{恢复后继续} (s2);
        \draw[arr] (s2) -- node[right,scale=0.72]{恢复后返回} (s3);
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.44\linewidth}
      \begin{block}{Coroutine frame 保存什么}
        \begin{itemize}
          \item 当前状态编号和恢复入口；
          \item 跨挂起点仍要使用的局部变量；
          \item promise / future / task 等返回协议对象；
          \item 异常、取消和最终挂起所需的元数据。
        \end{itemize}
      \end{block}
      \amberpoint{关键边界}{无栈协程只能在被标记的 coroutine / async 函数内部挂起，因此会产生“函数颜色”问题。}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{有栈协程：保存的是调用栈}
  \begin{columns}[T,totalwidth=\linewidth]
    \begin{column}{0.47\linewidth}
      \begin{block}{状态保存方式}
        \begin{itemize}
          \item 每个任务有可保存的用户态栈；
          \item 挂起时保存栈指针、寄存器和调度状态；
          \item 恢复时把栈重新装载到运行线程上。
        \end{itemize}
      \end{block}
      \tealpoint{优势}{可以在较深调用链中挂起，更接近“普通同步代码”的写法。}
      \amberpoint{代价}{运行时必须处理栈增长、栈复制、调度公平性和阻塞点识别。}
    \end{column}
    \begin{column}{0.48\linewidth}
      \centering
      \begin{tikzpicture}[
        box/.style={draw=lineGray, fill=softBg, rounded corners=2pt, minimum width=2.6cm, minimum height=0.72cm, align=center},
        arr/.style={-{Stealth[length=2.2mm]}, thick, draw=accentTeal}
      ]
        \node[box] (g) {\bfseries goroutine / virtual thread};
        \node[box,below=0.6cm of g] (stack) {可增长栈 / 可卸载栈帧};
        \node[box,below=0.6cm of stack] (carrier) {少量 OS carrier 线程};
        \node[box,below=0.6cm of carrier] (cpu) {CPU 核心};
        \draw[arr] (g) -- node[right,scale=0.72]{挂起时保存} (stack);
        \draw[arr] (stack) -- node[right,scale=0.72]{就绪时装载} (carrier);
        \draw[arr] (carrier) -- node[right,scale=0.72]{内核调度} (cpu);
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{三层协同：编译器 × 运行时 × OS}
  \centering
  \begin{tikzpicture}[
    box/.style={draw=lineGray, fill=softBg, rounded corners=2pt, minimum width=3.15cm, minimum height=1.05cm, align=center},

  \vspace{0.45cm}
  \begin{alertblock}{容易误解的一点}
    协程不是脱离操作系统的魔法。它必须依赖语言运行时识别“可等待事件”，再由 OS I/O 多路复用机制把“就绪”消息送回来。
  \end{alertblock}
\end{frame}

\section{语言：主流路线如何取舍}

\begin{frame}{主流语言：两条实现路线}
  \scriptsize
  \begin{tabularx}{\linewidth}{l l l Y}
    \toprule
    语言 & 类型 & 调度器 & 一句话特点 \\
    \midrule
    Go & 有栈 & GMP 内建 & \texttt{go} + channel，普通函数即可并发，网络轮询和调度器一体化。 \\
    Java 21 & 有栈 & JVM 调度 & 虚拟线程让同步代码低成本迁移，但要注意 pinned carrier 等阻塞边界。 \\
    C++20 & 无栈 & 标准不提供 & 语言只定义 \texttt{co\_await} 协议，Task、executor、I/O 后端由库决定。 \\
    Python & 无栈 & 单线程事件循环 & \texttt{async/await} 上手快；同步阻塞调用会卡住整个 loop，CPU 受 GIL 限制。 \\
    C\# & 无栈 & 线程池 / 上下文 & async/await 工程化成熟，异常传播和同步上下文规则需要理解。 \\
    Rust & 无栈 & Tokio 等运行时 & future 惰性执行、零成本抽象强，但生命周期和 Pin 增加学习门槛。 \\
    \bottomrule
  \end{tabularx}
  \vspace{0.25cm}
  \takeaway{“有没有协程”不是最终答案；调度器、标准库阻塞点覆盖和诊断工具决定它能不能在真实系统里用好。}
\end{frame}

      \tealpoint{Java virtual thread}{JVM 在多数阻塞 I/O 处卸载虚拟线程栈帧，保留同步代码风格。}
      \amberpoint{共同边界}{阻塞点若不能被识别，或长期持有锁/本地调用，轻量任务仍可能压住 carrier。}

  \end{columns}

\begin{frame}{C++ / Python / Rust：无栈路线更显式}
  \small
  \begin{columns}[T,totalwidth=\linewidth]
    \begin{column}{0.48\linewidth}
      \begin{block}{C++20 的低层特征}
        \begin{itemize}
          \item \texttt{promise\_type} 定义返回对象和挂起策略；
          \item \texttt{await\_ready/suspend/resume} 定义等待协议；
          \item \texttt{coroutine\_handle} 是恢复和销毁 frame 的句柄；
          \item 标准不规定 executor、Task 或异步 I/O。
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\linewidth}
      \begin{exampleblock}{Python / Rust 的工程启示}
        \begin{itemize}
          \item Python asyncio 简洁，但必须避免同步阻塞污染事件循环；
          \item Rust future 默认惰性，必须由运行时 poll 推进；
          \item 无栈路线通常内存更省，但 API 会明显区分同步与异步。
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
  \vspace{0.15cm}
  \takeaway{无栈协程把“何时挂起、谁来恢复、frame 何时销毁”暴露得更清楚：控制力更强，工程约束也更多。}
\end{frame}

\section{实验：本机结果说明什么}

\begin{frame}{I/O 密集型：协程快 7 到 23 倍}
  \vspace{0.08cm}
  {\scriptsize\color{muted}实验：10000 个逻辑任务，每个 sleep 100ms；5 次重复；平台为 i9-13900H + Windows 11。}
  \begin{columns}[T,totalwidth=\linewidth]
    \begin{column}{0.62\linewidth}
      \includegraphics[width=0.84\linewidth]{io_throughput.png}
    \end{column}
    \begin{column}{0.34\linewidth}
      \keypoint{Python}{threading 3.130s\\asyncio 0.446s\\约 7.0$\times$}
      \tealpoint{Go}{goroutine 0.116s\\运行时轻量任务优势明显}
      \amberpoint{Java}{platform 5.458s\\virtual 0.235s\\约 23.2$\times$}
    \end{column}
  \end{columns}
  \vspace{-0.05cm}
  \takeaway{优势来自等待期间释放执行权，而不是把 sleep “算得更快”。}
\end{frame}

\begin{frame}{CPU 密集与切换开销：协程不加速计算}
  \begin{columns}[T,totalwidth=\linewidth]
    \begin{column}{0.50\linewidth}
      \includegraphics[width=0.93\linewidth]{cpu_benchmark.png}
      \vspace{0.05cm}

    \end{column}
    \begin{column}{0.46\linewidth}
      \includegraphics[width=0.93\linewidth]{switch_overhead.png}
      \vspace{0.05cm}
      {\scriptsize\color{muted}切换测试并不同构：Python 为 1M 次 yield，Go/Java 为 100k 次 channel 或 sleep(0)。}

  \end{columns}
  \vspace{-0.02cm}
  \begin{alertblock}{解释边界}
    协程能减少等待占用的线程资源，但不能减少斐波那契递归本身的指令数；CPU 瓶颈仍应优先考虑并行计算、SIMD 或算法优化。
  \end{alertblock}
\end{frame}

\end{frame}

\section{选型：什么时候该用协程}

\begin{frame}{工程上真正难的是生命周期}
  \begin{columns}[T,totalwidth=\linewidth]
    \begin{column}{0.48\linewidth}
      \begin{block}{常见失败模式}
        \begin{itemize}
          \item 忘记取消子任务，形成协程泄漏；
          \item 超时只取消外层 future，底层 I/O 仍在跑；
          \item 同步阻塞函数进入事件循环；
          \item 错误在后台任务中被吞掉，调用方看不到失败。
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\linewidth}
      \begin{exampleblock}{更稳的实践}
        \begin{itemize}
          \item 用作用域管理任务组和取消传播；
          \item 为每个等待点设置超时与资源释放路径；
          \item 对阻塞库做隔离：线程池、异步替代或运行时适配；
          \item 用 trace、dump、profile 观察协程堆积位置。
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
  \takeaway{协程的核心工程问题不是“创建任务有多快”，而是任务失败后谁负责收尾。}
\end{frame}

\begin{frame}{三个问题快速判断}
  \begin{enumerate}
    \item \textbf{瓶颈是否真的在 I/O 等待？}\\
    {\small\color{muted}如果瓶颈在 CPU，优先考虑多线程、多进程、SIMD、GPU 或算法优化。}
    \vspace{0.25cm}
    \item \textbf{阻塞 API 能否被运行时接管或替换？}\\
    {\small\color{muted}Go / Java 21 更适合保留同步写法；C++ / Python / Rust 往往需要显式 async 化。}
    \vspace{0.25cm}
    \item \textbf{错误、取消、超时和背压能否结构化管理？}\\
    {\small\color{muted}没有这层治理，协程越轻量，系统越容易积累“看不见的后台任务”。}
  \end{enumerate}
  \vspace{0.15cm}
  \takeaway{三问都能肯定回答时，协程通常比传统线程池更适合高并发 I/O。}
\end{frame}

\begin{frame}{结论}
  \begin{block}{结论 1：协程承载大规模 I/O 并发}
    它降低的是“等待”的代价，不是“计算”的代价。
  \end{block}
  \begin{block}{结论 2：协程需要三层系统协同}
    编译器/解释器、语言运行时、OS I/O 多路复用共同完成挂起与恢复。
  \end{block}
  \begin{block}{结论 3：选型要看调度器与生态}
    Go / Java 21 强调透明迁移；C++ / Python / C\# / Rust 强调精细控制，各有场景。
  \end{block}
\end{frame}
